function calculateSample() { const design = document.getElementById('design').value; const alpha = parseFloat(document.getElementById('alpha').value); const zAlpha = alpha === 0.01 ? 2.576 : 1.96; let nResultText = ""; let strobeText = ""; if (design === 'cross') { const p = parseLocaleFloat('prevalence'); const d = parseLocaleFloat('precision'); if (isNaN(p) || isNaN(d)) { alert("Пожалуйста, введите корректные числа."); return; } const n = Math.ceil((Math.pow(zAlpha, 2) * p * (1 - p)) / Math.pow(d, 2)); nResultText = `🎯 Минимальный объем выборки: ${n} человек.`; strobeText = `Расчет минимального объема выборки выполнен в соответствии с руководством STROBE (Элемент 10) на основе классической формулы Кохрана (Cochran, 1977) для оценки долей в одномоментных исследованиях. Расчет проводился при априорной ожидаемой распространенности признака p=${(p*100).toFixed(1)}%, предельной погрешности d=${(d*100).toFixed(1)}% и уровне ошибки I рода alpha=${alpha} (95% ДИ).`; } else { const p0 = parseLocaleFloat('p0'); const effect = parseLocaleFloat('effectSize'); const powerVal = document.getElementById('power').value; if (isNaN(p0) || isNaN(effect)) { alert("Пожалуйста, введите корректные числовые параметры."); return; } const zBeta = powerVal === "0.90" ? 1.282 : 0.842; let p1 = 0; if (design === 'case-control') { p1 = (effect * p0) / (1 - p0 + (effect * p0)); } else { p1 = p0 * effect; } if (p1 >= 1 || p1 <= 0 || isNaN(p1)) { alert("Ошибка: При заданных параметрах математически невозможно рассчитать выборку. Проверьте размер эффекта (OR/RR) или p0."); return; } const pBar = (p0 + p1) / 2; const numerator = Math.pow(zAlpha * Math.sqrt(2 * pBar * (1 - pBar)) + zBeta * Math.sqrt(p0 * (1 - p0) + p1 * (1 - p1)), 2); const denominator = Math.pow(p1 - p0, 2); const nPerGroup = Math.ceil(numerator / denominator); const designName = design === 'case-control' ? "«случай-контроль»" : "когортного"; const effectName = design === 'case-control' ? `Отношении Шансов (OR) = ${effect}` : `Относительном Риске (RR) = ${effect}`; const powerPercent = powerVal === "0.90" ? "90%" : "80%"; nResultText = `🎯 Объем выборки: по ${nPerGroup} человек в КАЖДОЙ группе (Всего: ${nPerGroup * 2} участников).`; strobeText = `В рамках соблюдения критериев STROBE (Элемент 10: Объем выборки) для аналитического ${designName} исследования расчет объема групп выполнен по каноническому методу Флейса (Fleiss, 1980) и Келси (Kelsey, 1986) для сравнения двух независимых пропорций. При уровне ошибки I рода (alpha) ${alpha} и целевой статистической мощности (1-beta) ${powerPercent}, а также при базовой частоте p0=${(p0*100).toFixed(1)}% в группе сравнения и ожидаемом ${effectName}, минимально необходимый объем составил ${nPerGroup} участников в основной группе и ${nPerGroup} участников в группе контроля.`; } document.getElementById('calcOutcome').innerHTML = nResultText; document.getElementById('strobeOutput').innerHTML = strobeText; document.getElementById('resultBlock').style.display = 'block'; }
